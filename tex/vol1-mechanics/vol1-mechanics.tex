\chapter{Vectors} \label{ch:vectors}

\begin{center}
    \begin{tabular}{ c|c }
    \textbf{Vector}                                    & \textbf{Scalar} \\
    \hline
    Has both \textbf{direction} and \textbf{magnitude} & Has only magnitude \\
    Displacement                                       & Length/distance \\
    Velocity                                           & Speed \\
    Gravitational fields                               & Mass
    \end{tabular}

\section{Notation}

2D:
\begin{equation}
    \vec{v}
    = \boldsymbol{v}
    = \langle v_x, v_y \rangle
    = \left( v_x, v_y \right)
    = v_x \boldsymbol{\hat{\imath}} + v_y \boldsymbol{\hat{\jmath}}
\end{equation}

3D:
\begin{equation}
    \vec{v}
    = \boldsymbol{v}
    = \langle v_x, v_y, v_z \rangle
    = \left( v_x, v_y, v_z \right)
    = v_x \boldsymbol{\hat{\imath}} + v_y \boldsymbol{\hat{\jmath}} + v_z \boldsymbol{\hat{k}}
\end{equation}

Note that \(\vec{v}\) is more common in handwriting,
and \(\boldsymbol{v}\) is more common in print.

See \nameref{sec:unit-vectors} for information about \(\boldsymbol{\hat{\imath}}\),
\(\boldsymbol{\hat{\jmath}}\), and \(\boldsymbol{\hat{k}}\).

\section{Magnitude}

The size of the vector.

2D:
\begin{equation}
    \left\|\boldsymbol{v}\right\| = \sqrt{{v_x}^2 + {v_y}^2}
\end{equation}

3D:
\begin{equation}
    \left\|\boldsymbol{v}\right\| = \sqrt{{v_x}^2 + {v_y}^2 + {v_z}^2}
\end{equation}

\section{Unit Vectors} \label{sec:unit-vectors}

\textbf{Units vectors} are vectors with a magnitude of 1.
They are denoted using a "hat", e.g., \(\hat{v}\) is "v-hat".

The \textbf{standard unit vectors} are
\begin{equation}
    \begin{aligned}
        \boldsymbol{\hat{\imath}} &= \boldsymbol{\hat{x}} = \langle 1, 0, 0 \rangle, \\
        \boldsymbol{\hat{\jmath}} &= \boldsymbol{\hat{y}} = \langle 0, 1, 0 \rangle, \\
        \boldsymbol{\hat{z}} &= \boldsymbol{\hat{z}} = \langle 0, 0, 1 \rangle
    \end{aligned}
\end{equation}

\section{Operations}

Vector addition:
\begin{equation}
    \boldsymbol{u} + \boldsymbol{v} = \langle u_x + v_x, u_y + v_y, u_z + v_z \rangle
\end{equation}

Scalar multiplication:
\begin{equation}
    c \boldsymbol{v} = \langle c v_x, c v_y, c v_z \rangle
\end{equation}

Dot product:
\begin{equation}
    \begin{aligned}
        \boldsymbol{u} \cdot \boldsymbol{v} &= u_x v_x + u_y v_y + u_z v_z \\
        &= \left\|\boldsymbol{u}\right\| \left\|\boldsymbol{v}\right\| \cos\theta
    \end{aligned}
\end{equation}

Cross product:
\begin{equation}
    \begin{aligned}
        \boldsymbol{u} \cdot \boldsymbol{v}
        &= \langle u_y v_z - u_z v_y, u_z v_a - u_a v_z, u_a v_y - u_y v_a\rangle \\
        &= \left\|\boldsymbol{u}\right\| \left\|\boldsymbol{v}\right\| \boldsymbol{\hat{w}} \cos\theta \\
        &= \boldsymbol{w}
    \end{aligned}
\end{equation}
where \(\boldsymbol{\hat{w}}\) is a unit vector
orthogonal to both \(\boldsymbol{u}\) and \(\boldsymbol{v}\)
as given by the \textbf{Right Hand Rule}.
Hold your index, middle, and thumb fingers
such that all three are perpendicular to each other.
Your index finger represents \(\boldsymbol{u}\),
your middle finger \(\boldsymbol{v}\),
and your thumb \(\boldsymbol{w}\).
