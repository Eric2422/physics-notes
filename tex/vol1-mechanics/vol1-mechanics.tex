\section{Notation}

2D:
\begin{equation}
    \vec{v}
    = \boldsymbol{v}
    = \langle v_x, v_y \rangle
    = \left( v_x, v_y \right)
    = v_x \hat{i} + v_y \hat{j}
\end{equation}

3D:
\begin{equation}
    \vec{v}
    = \boldsymbol{v}
    = \langle v_x, v_y, v_z \rangle
    = \left( v_x, v_y, v_z \right)
    = v_x \hat{i} + v_y \hat{j} + v_z \hat{k}
\end{equation}

Note that \(\vec{v}\) is more common in handwriting,
and \(\boldsymbol{v}\) is more common in print.

See \nameref{sec:unit-vectors} for information about \(\hat{i}\), \(\hat{j}\),
and \(\hat{k}\).

\section{Magnitude}

The size of the vector.

2D:
\begin{equation}
    \left\|\boldsymbol{v}\right\| = \sqrt{{v_x}^2 + {v_y}^2}
\end{equation}

3D:
\begin{equation}
    \left\|\boldsymbol{v}\right\| = \sqrt{{v_x}^2 + {v_y}^2 + {v_z}^2}
\end{equation}
